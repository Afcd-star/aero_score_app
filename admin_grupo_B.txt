class CoachInfo {   // ✅ Email opcional (privado)

  const CoachInfo({
    required this.id,
    required this.firstName,
    required this.lastName,
    required this.username,
    this.email,
  });
  final String id;
  final String firstName;
  final String lastName;
  final String username; // ✅ Username en lugar de email
  final String? email;
  
  String get displayName {
    final fullName = '$firstName $lastName'.trim();
    return fullName.isNotEmpty ? fullName : username;
  }
  
  String get initials {
    final first = firstName.isNotEmpty ? firstName[0] : '';
    final last = lastName.isNotEmpty ? lastName[0] : '';
    return '$first$last'.toUpperCase();
  }
}

final coachesProvider = FutureProvider<List<CoachInfo>>((ref) async {
  debugPrint('🔍 [CoachProvider] ========== INICIANDO CARGA ==========');

  final pb = ref.watch(pocketBaseClientProvider);
  debugPrint('🔗 [CoachProvider] PocketBase URL: ${pb.baseURL}');

  try {
    debugPrint('📡 [CoachProvider] Filtrando: role = "Coach" && status = "Active"');
    
    final result = await pb.collection('users').getList(
      filter: 'role = "Coach" && status = "Active"',
      sort: 'firstName',
      perPage: 100,
    );
    
    debugPrint('✅ [CoachProvider] Total encontrados: ${result.items.length}');
    
    final coaches = <CoachInfo>[];

    for (final record in result.items) {
      final coach = CoachInfo(
        id: record.id,
        firstName: record.getStringValue('firstName'),
        lastName: record.getStringValue('lastName'),
        username: record.getStringValue('username'), // ✅ Siempre visible
        email: record.getStringValue('email'),        // ✅ Puede estar vacío/oculto
      );
      
      coaches.add(coach);
      debugPrint('   ➕ Coach: ${coach.displayName} (@${coach.username})');
    }

    debugPrint('✅ [CoachProvider] Total procesados: ${coaches.length}');
    return coaches;
    
  } catch (e, stack) {
    debugPrint('❌ [CoachProvider] Error: $e');
    debugPrint('📋 Stack: $stack');
    rethrow;
  }
});

/// Estado de la gestión de roles y usuarios.
class RoleManagementState {
  /// Crea el estado inicial de gestión de roles.
  const RoleManagementState({
    this.users = const AsyncValue.loading(),
    this.searchQuery = '',
    this.isUpdating = false,
  });

  /// Usuarios cargados o error de carga.
  final AsyncValue<List<UserEntity>> users;

  /// Texto actual del buscador.
  final String searchQuery;

  /// Indica si hay una actualización en curso.
  final bool isUpdating;

  /// Copia el estado con valores opcionales sobrescritos.
  RoleManagementState copyWith({
    AsyncValue<List<UserEntity>>? users,
    String? searchQuery,
    bool? isUpdating,
  }) {
    return RoleManagementState(
      users: users ?? this.users,
      searchQuery: searchQuery ?? this.searchQuery,
      isUpdating: isUpdating ?? this.isUpdating,
    );
  }
}

/// Notificador encargado de gestionar usuarios y roles.
class RoleManagementNotifier extends Notifier<RoleManagementState> {
  /// Crea el notificador sin dependencias externas directas.
  RoleManagementNotifier();

  @override
  RoleManagementState build() {
    return const RoleManagementState();
  }

  /// Expone el estado actual para lectura controlada.
  RoleManagementState get currentState => state;

  UserRepository get _repository => ref.read(userRepositoryProvider);

  /// Carga la lista de usuarios desde el repositorio.
  Future<void> loadUsers({String? search}) async {
    final query = search ?? state.searchQuery;
    state = state.copyWith(users: const AsyncValue.loading());

    final result = await _repository.getAllUsers(search: query);

    result.fold(
      (failure) {
        state = state.copyWith(
          users: AsyncValue.error(
            failure.message,
            StackTrace.current,
          ),
        );
      },
      (data) {
        state = state.copyWith(users: AsyncValue.data(data));
      },
    );
  }

  /// Actualiza el término de búsqueda y recarga usuarios.
  void updateSearchQuery(String query) {
    state = state.copyWith(searchQuery: query);
    unawaited(loadUsers(search: query));
  }

  /// Cambia el rol de un usuario y recarga si la operación es correcta.
  Future<bool> updateUserRole(
    String userId,
    UserRole newRole,
  ) async {
    state = state.copyWith(isUpdating: true);

    final result = await _repository.updateUserRole(
      userId: userId,
      newRole: newRole,
    );

    final success = result.fold(
      (failure) {
        state = state.copyWith(isUpdating: false);
        return false;
      },
      (_) {
        state = state.copyWith(isUpdating: false);
        return true;
      },
    );

    if (success) {
      unawaited(loadUsers());
    }

    return success;
  }

  /// Cambia el estado de un usuario y recarga si la operación es correcta.
  Future<bool> updateUserStatus(
    String userId,
    String newStatus,
  ) async {
    state = state.copyWith(isUpdating: true);

    final result = await _repository.updateUserStatus(
      userId: userId,
      newStatus: newStatus,
    );

    final success = result.fold(
      (failure) {
        state = state.copyWith(isUpdating: false);
        return false;
      },
      (_) {
        state = state.copyWith(isUpdating: false);
        return true;
      },
    );

    if (success) {
      unawaited(loadUsers());
    }

    return success;
  }
}

                    icon: const Icon(Icons.close, color: Colors.white, size: 24),
                    onPressed: () => Navigator.pop(context),
                    padding: EdgeInsets.zero,
                    constraints: const BoxConstraints(),
                  ),
                ),
              ],
            ),
          ),

          // ✅ BODY: Lista de competidores
          Expanded(
            child: competitorsAsync.when(
              loading: () => const Center(
                child: Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: [
                    CircularProgressIndicator(),
                    SizedBox(height: 16),
                    Text('Cargando competidores...'),
                  ],
                ),
              ),
              error: (err, _) => Center(
                child: Padding(
                  padding: const EdgeInsets.all(20),
                  child: Column(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      const Icon(
                        Icons.error_outline,
                        size: 48,
                        color: AppColors.destructiveLight,
                      ),
                      const SizedBox(height: 12),
                      Text(
                        'Error al cargar',
                        style: AppTextStyles.baseTextTheme.titleMedium?.copyWith(
                          color: AppColors.destructiveLight,
                        ),
                      ),
                      const SizedBox(height: 8),
                      Text(
                        err.toString(),
                        style: AppTextStyles.baseTextTheme.bodySmall,
                        textAlign: TextAlign.center,
                      ),
                    ],
                  ),
                ),
              ),
              data: (allCompetitors) {
                // Filtrar solo los competidores de este equipo
                final teamCompetitors = allCompetitors
                    .where((c) => competitorIds.contains(c.id))
                    .toList();

                if (teamCompetitors.isEmpty) {
                  return const Center(
                    child: Column(
                      mainAxisAlignment: MainAxisAlignment.center,
                      children: [
                        Icon(
                          Icons.people_outline,
                          size: 64,
                          color: Colors.grey,
                        ),
                        SizedBox(height: 16),
                        Text('Este equipo no tiene competidores'),
                      ],
                    ),
                  );
                }

                return ListView.separated(
                  padding: const EdgeInsets.all(20),
                  itemCount: teamCompetitors.length,
                  separatorBuilder: (_, _) => const SizedBox(height: 12),
                  itemBuilder: (context, index) {
                    final competitor = teamCompetitors[index];
                    return _buildCompetitorTile(competitor, isDark);
                  },
                );
              },
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildCompetitorTile(CompetitorSelectionInfo competitor, bool isDark) {
    return Container(
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: isDark ? AppColors.cardDark : Colors.grey.shade50,
        borderRadius: BorderRadius.circular(12),
        border: Border.all(
          color: isDark ? AppColors.borderDark : Colors.grey.shade200,
        ),
      ),
      child: Row(
        children: [
          // Avatar
          Container(
            width: 48,
            height: 48,
            decoration: BoxDecoration(
              gradient: AppColors.getGlowGradient(),
              borderRadius: BorderRadius.circular(12),
            ),
            child: Center(
              child: Text(
                competitor.initials.isNotEmpty ? competitor.initials : '?',
                style: const TextStyle(
                  color: Colors.white,
                  fontWeight: FontWeight.bold,
                  fontSize: 16,
                ),
              ),
            ),
          ),
          const SizedBox(width: 12),
          // Info
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  competitor.fullName,
                  style: AppTextStyles.baseTextTheme.bodyMedium?.copyWith(
                    fontWeight: FontWeight.w600,
                  ),
                ),
                const SizedBox(height: 4),
                Text(
                  'CI: ${competitor.nationalId}',
                  style: AppTextStyles.baseTextTheme.bodySmall?.copyWith(
                    color: isDark ? AppColors.mutedForegroundDark : AppColors.mutedForegroundLight,
                  ),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
